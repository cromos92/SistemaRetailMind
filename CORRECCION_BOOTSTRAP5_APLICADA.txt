🔍 PROBLEMA IDENTIFICADO
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Síntoma:
├─ Modal se abre (flash)
├─ Modal se cierra inmediatamente
└─ No se puede ver el contenido

Causa:
├─ Tu sistema usa Bootstrap 5
├─ Pero los modales estaban con sintaxis de Bootstrap 4
└─ Incompatibilidad → Modal se cierra automáticamente


✅ SOLUCIÓN APLICADA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Archivos Corregidos:
├─ ✅ modales_edicion_producto.html
│   ├─ data-dismiss → data-bs-dismiss
│   ├─ class="close" → class="btn-close"
│   └─ role="document" eliminado
│
├─ ✅ edicion_productos.js
│   ├─ .modal('show') → new bootstrap.Modal().show()
│   └─ .modal('hide') → bootstrap.Modal.getInstance().hide()
│
└─ ✅ verGestionProductos.html
    └─ Fallback actualizado a Bootstrap 5


🎯 QUÉ CAMBIÓ
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Bootstrap 4 (ANTES):
├─ <button data-dismiss="modal">
├─ <button class="close">
├─ $('#modal').modal('show')
└─ $('#modal').modal('hide')

Bootstrap 5 (AHORA):
├─ <button data-bs-dismiss="modal">      ✅
├─ <button class="btn-close">            ✅
├─ new bootstrap.Modal(el).show()        ✅
└─ bootstrap.Modal.getInstance(el).hide() ✅
